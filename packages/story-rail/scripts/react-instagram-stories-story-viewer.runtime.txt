function IC_createDirectionAdapter(direction) {
  const isRtl = direction === "rtl";
  const nextOwnerSign = isRtl ? 1 : -1;
  return {
    itemActionForSide(side) {
      return side === (isRtl ? "left" : "right") ? "next" : "previous";
    },
    ownerActionForPhysicalDelta(deltaX) {
      if (deltaX === 0) return null;
      return Math.sign(deltaX) === nextOwnerSign ? "next" : "previous";
    },
    cubeTargetForOwner(ownerAction) {
      return ownerAction === "next" ? nextOwnerSign : -nextOwnerSign;
    },
    faceAngleForOwner(ownerAction) {
      return `${-this.cubeTargetForOwner(ownerAction) * 90}deg`;
    },
  };
}

function IC_prefersReducedMotion() {
  return typeof window !== "undefined" &&
    typeof window.matchMedia === "function" &&
    window.matchMedia("(prefers-reduced-motion: reduce)").matches;
}

function IC_waitForTransform(element, duration, callback) {
  let complete = false;
  let fallbackId;
  const finish = () => {
    if (complete) return;
    complete = true;
    element.removeEventListener("transitionend", onEnd);
    element.removeEventListener("transitioncancel", onCancel);
    if (fallbackId !== undefined) window.clearTimeout(fallbackId);
    callback();
  };
  const onEnd = (event) => {
    if (event.target === element && event.propertyName === "transform") finish();
  };
  const onCancel = (event) => {
    if (event.target === element) finish();
  };
  if (duration <= 0) requestAnimationFrame(finish);
  else {
    element.addEventListener("transitionend", onEnd);
    element.addEventListener("transitioncancel", onCancel);
    fallbackId = window.setTimeout(finish, duration + 96);
  }
  return () => {
    if (complete) return;
    complete = true;
    element.removeEventListener("transitionend", onEnd);
    element.removeEventListener("transitioncancel", onCancel);
    if (fallbackId !== undefined) window.clearTimeout(fallbackId);
  };
}

function IC_clearCubeStyles(wrapper) {
  if (!wrapper) return;
  wrapper.style.transition = "";
  wrapper.style.transform = "";
  wrapper.style.transformStyle = "";
  wrapper.style.removeProperty("--face-depth");
  if (wrapper.parentElement) wrapper.parentElement.style.perspective = "";
}

function IC_resetDismissStyles(container) {
  const viewport = container?.querySelector(".story-viewer-cube-viewport");
  const overlay = container?.querySelector(".story-viewer-overlay");
  if (viewport) {
    viewport.style.transition = "";
    viewport.style.transform = "";
    viewport.style.borderRadius = "";
    viewport.style.overflow = "";
    viewport.style.opacity = "";
  }
  if (overlay) {
    overlay.style.transition = "";
    overlay.style.opacity = "";
  }
}

function IC_lockDocument(scrollPositionRef) {
  const html = document.documentElement;
  const body = document.body;
  const scrollX = window.scrollX;
  const previous = {
    htmlOverflow: html.style.overflow,
    htmlOverscroll: html.style.overscrollBehavior,
    bodyOverflow: body.style.overflow,
    bodyOverscroll: body.style.overscrollBehavior,
  };
  scrollPositionRef.current = window.scrollY;
  html.style.overflow = "hidden";
  html.style.overscrollBehavior = "none";
  body.style.overflow = "hidden";
  body.style.overscrollBehavior = "none";
  return () => {
    html.style.overflow = previous.htmlOverflow;
    html.style.overscrollBehavior = previous.htmlOverscroll;
    body.style.overflow = previous.bodyOverflow;
    body.style.overscrollBehavior = previous.bodyOverscroll;
    window.scrollTo(scrollX, scrollPositionRef.current);
  };
}

function IC_clampOwnerDrag(rawPercentage, ownerAction, currentOwner, ownerCount) {
  let percentage = Math.max(-0.95, Math.min(0.95, rawPercentage));
  const atBoundary =
    (ownerAction === "previous" && currentOwner === 0) ||
    (ownerAction === "next" && currentOwner === ownerCount - 1);
  if (atBoundary) percentage *= 0.22;
  return percentage;
}

function IC_decideOwnerCommit(percentage, velocity, direction, currentOwner, ownerCount) {
  const ownerAction = Math.abs(velocity) >= 0.55
    ? direction.ownerActionForPhysicalDelta(velocity)
    : direction.ownerActionForPhysicalDelta(percentage);
  const atBoundary =
    (ownerAction === "previous" && currentOwner === 0) ||
    (ownerAction === "next" && currentOwner === ownerCount - 1);
  return {
    ownerAction,
    shouldCommit: !!ownerAction && !atBoundary &&
      (Math.abs(percentage) >= 0.28 || Math.abs(velocity) >= 0.55),
  };
}

    }
  }, [isOpen]);

  REACT.useEffect(() => {
    if (!isOpen) return;
    if (initialNotificationRef.current) {
      initialNotificationRef.current = false;
      onStoryChange?.(currentUserIndex, currentStoryIndex);
      return;
    }
    if (isQueryParamMode && currentUser && currentStory &&
        (searchParams.get("user") !== currentUser.id || searchParams.get("story") !== currentStory.id)) {
      setSearchParams({ user: currentUser.id, story: currentStory.id }, { replace: true });
    }
    onStoryChange?.(currentUserIndex, currentStoryIndex);
  }, [currentUserIndex, currentStoryIndex, onStoryChange, isOpen, isQueryParamMode, searchParams, setSearchParams]);

  const onLoadError = REACT.useCallback(() => {
    console.warn("Story item failed to load, skipping...");
    window.setTimeout(next, 500);
  }, [next]);

  if (!isOpen || !currentUser || !currentStory) return null;
  const previousUserIndex = cubeState?.leftUserIndex;
  const nextUserIndex = cubeState?.rightUserIndex;
  const previousUser = previousUserIndex === undefined ? null : users[previousUserIndex];
  const nextUser = nextUserIndex === undefined ? null : users[nextUserIndex];
  const previousStoryIndex = previousUserIndex !== undefined && previousUserIndex < currentUserIndex
    ? lastViewedStoryRef.current.get(previousUserIndex) ?? 0
    : currentStoryIndex;
  const nextStoryIndex = nextUserIndex !== undefined && nextUserIndex > currentUserIndex
    ? lastViewedStoryRef.current.get(nextUserIndex) ?? 0
    : currentStoryIndex;

  const renderFace = (faceUser, faceStory, faceStoryIndex, faceProgress, isLive) => JSS(JSF, {
    children: [
      JSS("div", {
        className: x("story-viewer-header", classNames?.header),
        dir,
        children: [
          JSX(Ne, {
            total: faceUser.stories.length,
            currentIndex: faceStoryIndex,
            progress: faceProgress,
            classNames: classNames?.progressBars,
            labels,
          }),
          JSS("div", {
            className: x("story-viewer-user-info", classNames?.userInfo),
            dir,
            children: [
              JSX("img", {
                src: faceUser.avatarUrl,
                alt: `${faceUser.username} avatar`,
                className: x("story-viewer-avatar", classNames?.avatar),
              }),
              JSX("span", {
                className: x("story-viewer-username", classNames?.username),
                children: faceUser.username,
              }),
            ],
          }),
          JSX("button", {
            className: x("story-viewer-close", classNames?.closeButton),
            onClick: close,
            "aria-label": labels.close ?? "Close story viewer",
            type: "button",
            children: "×",
          }),
        ],
      }, faceUser.id),
      JSX("div", {
        className: x("story-viewer-items", classNames?.items),
        dir,
        children: JSX(ke, {
          item: faceStory,
          isActive: isLive,
          isPaused: isLive ? isPaused || transitionStateRef.current !== "idle" : true,
          onDurationDetected: isLive ? (duration) => timer.setDuration(duration) : undefined,
          onLoadError: isLive ? onLoadError : undefined,
          onBufferingChange: isLive ? onBufferingChange : undefined,
          controls,
          classNames: classNames?.storyItem,
        }),
      }),
      isLive && JSS("div", {
        className: "story-viewer-nav-hints",
        dir,
        children: [
          JSX("div", { className: "story-viewer-nav-hint story-viewer-nav-hint-left" }),
          JSX("div", { className: "story-viewer-nav-hint story-viewer-nav-hint-right" }),
        ],
      }),
    ],
  });

  const isCube = !!(cubeState && previousUser && nextUser);
  const depth = isCube
    ? (faceWidthRef.current || Math.min(typeof window !== "undefined" ? window.innerWidth : 500, 500)) / 2
    : 0;
  const faceStyle = (angle) => ({
    position: "absolute",
    top: 0,
    left: 0,
    width: "100%",
    height: "100%",
    background: "#000",
    display: "flex",
    flexDirection: "column",
    overflow: "hidden",
    backfaceVisibility: "hidden",
    transform: `rotateY(${angle}) translateZ(${depth}px)`,
  });

  const content = JSS("div", {
    ref: containerRef,
    className: x("story-viewer", classNames?.root),
    dir,
    "data-transition-state": transitionState,
    role: "dialog",
    "aria-modal": "true",
    "aria-label": labels.dialog?.(currentUser.username || "user") ??
      `Stories by ${currentUser.username || "user"}`,
    "aria-describedby": "story-viewer-description",
    onPointerDown,
    onPointerMove,
    onPointerUp,
    onPointerCancel: cancelGesture,
    onLostPointerCapture: cancelGesture,
    children: [
      JSX("div", { className: x("story-viewer-overlay", classNames?.overlay), onClick: close }),
      JSX("div", {
        id: "story-viewer-description",
        className: "sr-only",
        children: labels.description ??
          "Instagram-style stories viewer. Use either side for items, swipe horizontally for people, swipe down to close, and hold to pause.",
      }),
      JSX("div", {
        "aria-live": "polite",
        "aria-atomic": "true",
        className: "sr-only",
        children: labels.announcement?.(
          currentStoryIndex + 1,
          totalStories,
          currentUser.username || "user",
        ) ?? `Viewing story ${currentStoryIndex + 1} of ${totalStories} by ${currentUser.username}`,
      }),
      JSX("div", {
        className: "story-viewer-cube-viewport",
        dir,
        style: isCube ? { perspective: `${depth * 16}px` } : undefined,
        children: JSS("div", {
          ref: cubeWrapperRef,
          className: "story-viewer-cube-wrapper",
          dir,
          style: isCube ? { transformStyle: "preserve-3d", transform: re(depth, 0) } : undefined,
          children: [
            JSX("div", {
              ref: focusTrapRef,
              className: x("story-viewer-content", classNames?.content),
              dir,
              style: isCube ? {
                willChange: "auto",
                backfaceVisibility: "hidden",
                transform: `translateZ(${depth}px)`,
              } : undefined,
              children: renderFace(
                currentUser,
                currentStory,
                currentStoryIndex,
                externalProgress ?? timer.progress,
                !isCube,
              ),
            }),
            isCube && previousUser && JSX("div", {
              dir,
              "data-owner-face": "previous",
              style: faceStyle(direction.faceAngleForOwner("previous")),
              children: renderFace(
                previousUser,
                previousUser.stories[previousStoryIndex],
                previousStoryIndex,
                0,
                false,
              ),
            }),
            isCube && nextUser && JSX("div", {
              dir,
              "data-owner-face": "next",
              style: faceStyle(direction.faceAngleForOwner("next")),
              children: renderFace(
                nextUser,
                nextUser.stories[nextStoryIndex],
                nextStoryIndex,
                0,
                false,
              ),
            }),
          ],
        }),
      }),
    ],
  });
  return PORTAL(content, document.body);
});
